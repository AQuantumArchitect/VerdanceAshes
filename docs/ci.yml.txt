name: CI
on:
  pull_request:
  push:
    branches: [master]

jobs:
  hub-tests:
    runs-on: windows-latest
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - name: Hub tests
        run: python tests/run_tests.py

  ship-dry-run:
    runs-on: windows-latest
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      # The pinned Umwelt (tools/umwelt_pin.json) is a private repo, so CI cannot clone it yet.
      # Until a read-only deploy key exists, stage a stub Umwelt and let ship.ps1 drift on it.
      # This checks ship's plumbing and licence/manifest shape, not the real pin.
      - name: Stub Umwelt
        shell: pwsh
        run: |
          $u = Join-Path $env:RUNNER_TEMP "umwelt"
          New-Item -ItemType Directory -Force -Path (Join-Path $u "src\umwelt") | Out-Null
          Set-Content (Join-Path $u "src\umwelt\__init__.py") "# CI stub"
          Set-Content (Join-Path $u "LICENSE") "Apache License 2.0 (CI stub)"
          "UMWELT_SRC=$(Join-Path $u 'src')" | Add-Content $env:GITHUB_ENV
          Write-Host "::warning::Umwelt is a CI stub; the real pin is not verified here."
      - name: Ship dry run
        shell: pwsh
        run: ./scripts/ship.ps1 -SkipPython -SkipZip -AllowUmweltDrift
      - name: Check staged build
        shell: pwsh
        run: |
          $v = ((Get-Content -Raw VERSION).Trim() -split "[\r\n]+")[0].Trim()
          python tools/check_ship_dryrun_v1.py "export/dist/VerdanceAshes_v$v"
